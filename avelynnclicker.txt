Shinies
	*luckyheart
		on click:log Woop
		movement:onRight moveLeft fade bounce:0.05
		frequency:60
		frequency variation:30
		icon:stuff/luckycarrot.png
		class:bigButton
		on click:
			$boost=1
			if (have clover) $boost=4
			if (chance(25%))
				//get at least 7, or between 1 and 3 minutes of our heart production
				$amount=max(7,random(hearts:ps*60*1,hearts:ps*60*3))*$boost
				toast The lucky heart grants you<//><b>[$amount] hearts</b>.
				yield $amount hearts
			else
				$amount=1*$boost
				toast The lucky heart grants you<//><b>[$amount] golden heart[s?$amount]</b>!
				yield $amount goldenHeart
			end
		end

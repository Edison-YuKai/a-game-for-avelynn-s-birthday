Shinies
	*luckyheart
		on click:log Woop
		movement:anywhere wiggle grow
		frequency:60
		frequency variation:30
		duration:30
		icon:stuff/luckyBunny.png
		class:bigButton
		// must be 256 x 256
		on click:
			$boost=1
			if (have goldenUpgrade4) $boost=4
			if (chance(25%))
				//get 30 minutes of heart production
				$amount=max(random(hearts:ps*60*30))*$boost
				toast The lucky heart grants you 30 minutes of heart production:<//><b>[$amount] hearts</b>!
				yield $amount hearts
			else
				$amount=max(random(1,7))*$boost
				toast The lucky heart grants you<//><b>[$amount] golden heart[s?$amount]</b>!
				yield $amount goldenHeart
			end
		end
